--- SW-SERVER
ena
conf t
hostname SW-VTP_SERVER

vtp mode server
vtp version 2
vtp domain itis
vtp password cisco

interface fastEthernet 0/1
switchport trunk encapsulation dot1q
switchport mode trunk

vlan 10
name A
exit

vlan 20
name B
exit

vlan 30
name C
exit

no shutdown
exit

no shutdown
exit

no shutdown
